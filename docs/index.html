Quick Playwright / Python Integration

Subscribe to DOMPulse structured events without capturing a single screenshot:

from playwright.sync_api import sync_playwright

with sync_playwright() as p:
    browser = p.chromium.launch_persistent_context(
        user_data_dir="/tmp/agent-chrome",
        headless=False,
        args=["--load-extension=./dist"]
    )
    page = browser.new_page()
    page.goto("https://store.example.com")

    # Agent receives structured JSON events with screen coordinates in 15ms!
    page.evaluate("""
        window.addEventListener('dompulse:event', (e) => {
            console.log('UI Changed:', e.detail.type, e.detail.bbox);
        });
    """)
